BEGIN

    INPUT pendapatan
    INPUT jumlah_pengeluaran

    total_pengeluaran ← 0

    FOR i ← 1 TO jumlah_pengeluaran DO

        INPUT kategori
        INPUT jumlah

        total_pengeluaran ← total_pengeluaran + jumlah

        SIMPAN kategori dan jumlah

    END FOR

    sisa_saldo ← pendapatan - total_pengeluaran

    persentase_saldo ← (sisa_saldo / pendapatan) × 100

    IF persentase_saldo >= 30 THEN
        kondisi ← "Baik"

    ELSE IF persentase_saldo >= 10 THEN
        kondisi ← "Perlu Perhatian"

    ELSE
        kondisi ← "Kritis"

    END IF

    OUTPUT "===== RINGKASAN ANGGARAN ====="
    OUTPUT "Total Pendapatan: ", pendapatan
    OUTPUT "Daftar Pengeluaran:"

    FOR setiap data pengeluaran DO
        OUTPUT kategori, jumlah
    END FOR

    OUTPUT "Total Pengeluaran: ", total_pengeluaran
    OUTPUT "Sisa Saldo: ", sisa_saldo
    OUTPUT "Persentase Sisa Saldo: ", persentase_saldo, "%"
    OUTPUT "Kondisi Keuangan: ", kondisi

END
